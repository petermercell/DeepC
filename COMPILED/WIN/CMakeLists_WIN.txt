# ============================================================================
# Helper function
# ============================================================================
function(add_nuke_plugin PLUGIN_NAME)
    add_library(${PLUGIN_NAME} SHARED ${ARGN})

    # Static runtime for portability
    set_property(TARGET ${PLUGIN_NAME} PROPERTY
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

    target_include_directories(${PLUGIN_NAME} PRIVATE "${NDKDIR}/include")

    target_compile_definitions(${PLUGIN_NAME} PRIVATE
        _USE_MATH_DEFINES
        NOMINMAX
    )

    target_link_libraries(${PLUGIN_NAME} PRIVATE ${DDIMAGE_LIBRARY})

    set_target_properties(${PLUGIN_NAME} PROPERTIES
        PREFIX ""
        SUFFIX ".dll"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release
        LIBRARY_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    )
endfunction()
